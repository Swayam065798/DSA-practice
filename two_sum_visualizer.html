func twoSum(_ nums: [Int], _ target: Int) -> [Int] {
  var map: [Int: Int] = [:]
  for (index, num) in nums.enumerated() {
    let need = target - num
    if let foundIndex = map[need] {
      return [foundIndex, index]
    }
    map[num] = index
  }
  return []
}
